Transaction Reporting

Intermediate Records Compact constructor Optional Text blocks Java 21 or newer

The problem

The bank exports transactions as text. The program turns each line into a value that is always valid, reports per account, and answers questions where the honest answer is sometimes none.

A record gives you the fields, accessors, equals, hashCode and toString in one line. Its compact constructor validates, so an invalid one cannot exist. Optional puts "maybe nothing" in the type.

The whole program as one picture. Read it top to bottom; the colour of each card says what kind of step it is.

A transaction slip that cannot be filled in wrongly, and a lookup that is allowed to say none.

If you are experienced

Records are immutable, value-based carriers. Optional belongs on return types, not fields or parameters.

Text version of the flow
  1. A text block of lines (input): Several lines of CSV in one Java string.
  2. Parse each line (check): Transaction.parse returns Optional; bad lines are listed.
  3. Compact constructor validates (check): Amount must be positive; scale is fixed to two places.
  4. Ask the ledger (work): findById, largestDeposit, latestFor each return Optional.
  5. Account summaries (result): Deposits, withdrawals and net per account.

Read in this order

  1. 1Kind.java, AccountSummary.javaAn enum and a record with a computed method.
  2. 2Transaction.javaCompact constructor, static parse, withAmount.

The code

src/main/java/com/codechanakya/reporting/Transaction.java
package com.codechanakya.reporting;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Optional;

/**
 * One transaction, written as a record.
 *
 * The one line in the header gives you: private final fields, a constructor, accessors named
 * after the fields (id(), not getId()), and equals, hashCode and toString that use every field.
 * Everything below is extra: validation, a parser and a helper method.
 *
 * @param id      unique id such as T-1001
 * @param account the account it belongs to
 * @param kind    deposit or withdrawal
 * @param amount  always positive; the kind says which way the money moved
 * @param date    the day it happened
 */
public record Transaction(String id, String account, Kind kind, BigDecimal amount, LocalDate date)
        implements Comparable<Transaction> {

    /**
     * The compact constructor. It has no parameter list, runs before the fields are assigned, and
     * is the place to check values. The fields are assigned for you at the end, so do not write
     * this.amount = ... here (the compiler rejects it).
     */
    public Transaction {
        if (amount.signum() <= 0) {
            throw new IllegalArgumentException("Amount must be positive for " + id);
        }
        // Reassigning the PARAMETER is allowed, and is what gets stored. setScale(2) with no rounding
        // mode throws for an amount like 10.005, which is what we want: money has two decimals.
        amount = amount.setScale(2);
    }

    /** Positive for money in, negative for money out. */
    public BigDecimal signedAmount() {
        return kind == Kind.DEPOSIT ? amount : amount.negate();
    }

    /** A copy with a different amount. Records cannot change, so "changing" one means making another. */
    public Transaction withAmount(BigDecimal newAmount) {
        return new Transaction(id, account, kind, newAmount, date);
    }

    @Override
    public int compareTo(Transaction other) {
        return date.compareTo(other.date);
    }

    /**
     * Reads one CSV line: id,account,kind,amount,date.
     * Returns an empty Optional when the line cannot be used, instead of null or an exception.
     */
    public static Optional<Transaction> parse(String line) {
        String[] p = line.split(",");
        if (p.length != 5) {
            return Optional.empty();
        }
        try {
            return Optional.of(new Transaction(p[0].trim(), p[1].trim(), Kind.valueOf(p[2].trim()),
                    new BigDecimal(p[3].trim()), LocalDate.parse(p[4].trim())));
        } catch (RuntimeException e) {
            // bad kind, bad number, bad date or a non-positive amount: all mean "not a usable line"
            return Optional.empty();
        }
    }
}
src/main/java/com/codechanakya/reporting/Ledger.java
package com.codechanakya.reporting;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.TreeMap;

/** Holds transactions and answers questions about them. Several answers may be "nothing", hence Optional. */
public class Ledger {

    private final List<Transaction> transactions = new ArrayList<>();
    private final List<String> rejected = new ArrayList<>();

    /** Loads CSV text. Lines that do not parse are remembered, not fatal. */
    public void load(String csv) {
        int lineNumber = 0;
        for (String line : csv.lines().toList()) {          // String.lines() splits on any line ending
            lineNumber++;
            if (line.isBlank() || line.startsWith("id,")) {
                continue;
            }
            Optional<Transaction> parsed = Transaction.parse(line);
            if (parsed.isPresent()) {
                transactions.add(parsed.get());
            } else {
                rejected.add("line " + lineNumber + ": " + line);
            }
        }
    }

    public List<Transaction> all() {
        return List.copyOf(transactions);
    }

    public List<String> rejected() {
        return rejected;
    }

    /** The transaction with this id, if there is one. The caller is forced to think about "no". */
    public Optional<Transaction> findById(String id) {
        return transactions.stream().filter(t -> t.id().equals(id)).findFirst();
    }

    /** The largest single deposit, if there were any deposits at all. */
    public Optional<Transaction> largestDeposit() {
        return transactions.stream()
                .filter(t -> t.kind() == Kind.DEPOSIT)
                .max(Comparator.comparing(Transaction::amount));
    }

    /** The most recent transaction for an account, if the account has any. */
    public Optional<Transaction> latestFor(String account) {
        return transactions.stream()
                .filter(t -> t.account().equals(account))
                .max(Comparator.naturalOrder());           // uses Transaction.compareTo, which is by date
    }

    /** One row per account, accounts in alphabetical order. */
    public List<AccountSummary> summaries() {
        Map<String, List<Transaction>> byAccount = new TreeMap<>();
        for (Transaction t : transactions) {
            byAccount.computeIfAbsent(t.account(), key -> new ArrayList<>()).add(t);
        }
        List<AccountSummary> rows = new ArrayList<>();
        for (Map.Entry<String, List<Transaction>> entry : byAccount.entrySet()) {
            BigDecimal in = BigDecimal.ZERO.setScale(2);
            BigDecimal out = BigDecimal.ZERO.setScale(2);
            for (Transaction t : entry.getValue()) {
                if (t.kind() == Kind.DEPOSIT) {
                    in = in.add(t.amount());
                } else {
                    out = out.add(t.amount());
                }
            }
            rows.add(new AccountSummary(entry.getKey(), entry.getValue().size(), in, out));
        }
        return rows;
    }
}

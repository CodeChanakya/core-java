Banking Application

The problem

A small bank keeps accounts: an owner, a balance, and rules for deposits, withdrawals and transfers.

A class bundles the facts and the rules into one unit, and private fields stop anyone changing a balance without going through those rules.

Read in this order

  1. 1Account.javaPrivate fields, a checking constructor, deposit and withdraw.
  2. 2Bank.javaMany accounts, and the transfer rule that spans two.
  3. 3Main.javaA scripted day at the bank.

src/main/java/com/codechanakya/bank/Account.java
package com.codechanakya.bank;

import java.math.BigDecimal;
import java.math.RoundingMode;

/**
 * One customer's bank account.
 *
 * The balance is private on purpose. Code outside this class cannot set it,
 * so it only ever changes through deposit, withdraw or Bank.transfer, and each
 * of those checks its input first. Keeping the data behind a few checked
 * methods is what "encapsulation" means in practice.
 */
public class Account {

    private final String accountNumber;
    private final String holderName;
    private BigDecimal balance;

    /**
     * Creates an account. Refuses to build one that makes no sense.
     *
     * Throwing an exception is how a constructor says "I cannot create this
     * object". Phase 4 covers exceptions properly; for now, read it as
     * "stop here and tell the caller what was wrong".
     */
    public Account(String accountNumber, String holderName, BigDecimal openingBalance) {
        if (holderName == null || holderName.isBlank()) {
            throw new IllegalArgumentException("Account holder name is required");
        }
        if (openingBalance.signum() < 0) {
            throw new IllegalArgumentException("Opening balance cannot be negative");
        }
        this.accountNumber = accountNumber;
        this.holderName = holderName.trim();
        this.balance = openingBalance.setScale(2, RoundingMode.HALF_UP);
    }

    /** Adds money. Returns false, and changes nothing, if the amount is zero or negative. */
    public boolean deposit(BigDecimal amount) {
        if (amount.signum() <= 0) {
            return false;
        }
        balance = balance.add(amount.setScale(2, RoundingMode.HALF_UP));
        return true;
    }

    /**
     * Takes money out. Returns false, and changes nothing, if the amount is not
     * positive or is more than the balance. There is no overdraft in this bank.
     */
    public boolean withdraw(BigDecimal amount) {
        BigDecimal exact = amount.setScale(2, RoundingMode.HALF_UP);
        if (exact.signum() <= 0 || exact.compareTo(balance) > 0) {
            return false;
        }
        balance = balance.subtract(exact);
        return true;
    }

    public String getAccountNumber() {
        return accountNumber;
    }

    public String getHolderName() {
        return holderName;
    }

    public BigDecimal getBalance() {
        return balance;
    }

    @Override
    public String toString() {
        return String.format("%-9s %-14s %12s", accountNumber, holderName, balance.toPlainString());
    }
}
src/main/java/com/codechanakya/bank/Bank.java
package com.codechanakya.bank;

import java.math.BigDecimal;
import java.util.ArrayList;

/**
 * A bank is a collection of accounts plus the rules that involve more than one
 * of them, such as a transfer.
 *
 * ArrayList is a list that grows as you add to it. Phase 5 covers collections
 * properly; here, think of it as an array that never runs out of room.
 */
public class Bank {

    private final String name;
    private final ArrayList<Account> accounts = new ArrayList<>();
    private int nextNumber = 1001;

    public Bank(String name) {
        this.name = name;
    }

    /** Opens an account and gives it the next free number, such as ACC-1001. */
    public Account openAccount(String holderName, BigDecimal openingBalance) {
        Account account = new Account("ACC-" + nextNumber, holderName, openingBalance);
        nextNumber++;
        accounts.add(account);
        return account;
    }

    /** Returns the account with this number, or null when there is none. */
    public Account find(String accountNumber) {
        for (Account account : accounts) {
            if (account.getAccountNumber().equals(accountNumber)) {
                return account;
            }
        }
        return null;
    }

    /**
     * Moves money from one account to another.
     *
     * The withdrawal comes first. If it fails, nothing has changed and we stop.
     * The deposit cannot fail after that, because the amount has already been
     * accepted as positive, so no money can vanish halfway.
     */
    public boolean transfer(String fromNumber, String toNumber, BigDecimal amount) {
        Account from = find(fromNumber);
        Account to = find(toNumber);
        if (from == null || to == null || from == to) {
            return false;
        }
        if (!from.withdraw(amount)) {
            return false;
        }
        return to.deposit(amount);
    }

    /** The sum of every balance in the bank. */
    public BigDecimal totalHoldings() {
        BigDecimal total = BigDecimal.ZERO.setScale(2);
        for (Account account : accounts) {
            total = total.add(account.getBalance());
        }
        return total;
    }

    public String getName() {
        return name;
    }

    public int accountCount() {
        return accounts.size();
    }

    /** Prints one line per account and the total. */
    public void printStatement() {
        System.out.println("--- " + name + " ---");
        for (Account account : accounts) {
            System.out.println(account);
        }
        System.out.println("Total holdings: " + totalHoldings().toPlainString());
    }
}

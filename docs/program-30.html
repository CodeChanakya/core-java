Product Recommendation Filtering

Advanced Sealed types Pattern matching Record patterns Phase project Java 21 or newer

The problem

An online shop sells books, gadgets and groceries. Each shopper has a budget, a minimum rating, wanted kinds and maybe a dietary rule. The shop wants ordered recommendations with a reason for each, and when a new kind of product is added the compiler should point out every place to update.

This Phase 7 project combines records, sealed interfaces, exhaustive pattern-matching switches without a default, rules held as data, streams and Optional.

Text version of the flow
  1. Products as a sealed set (input): Book, Gadget or Grocery, nothing else.
  2. Wishes become one Rule (work): A tree of records: MinRating, MaxPrice, OfKind, Not, AllOf, AnyOf.
  3. Match with an exhaustive switch (check): Record patterns, no default: the compiler checks every case.
  4. Score and sort (work): Streams filter, score and order the matches.
  5. Ranked picks and a best pick (result): Each has a reason; Optional says none when nothing matches.

src/main/java/com/codechanakya/recommend/Product.java
package com.codechanakya.recommend;

import java.math.BigDecimal;

/**
 * Everything the shop sells is exactly one of three kinds.
 *
 * "sealed" with a "permits" list closes the family: no other class can implement Product.
 * That promise is what lets the compiler check, later, that a switch has covered every kind.
 */
public sealed interface Product permits Book, Gadget, Grocery {

    String name();

    BigDecimal price();

    double rating();
}
src/main/java/com/codechanakya/recommend/RuleEngine.java
package com.codechanakya.recommend;

import java.util.Locale;
import java.util.stream.Collectors;

/** Reads a Rule and answers questions about it. All the work is pattern-matching switches. */
public final class RuleEngine {

    private RuleEngine() {
    }

    /**
     * Does this product satisfy the rule?
     *
     * There is no "default" branch. Rule is sealed, so the compiler knows the seven cases below are
     * all of them, and adding an eighth rule type will stop this method compiling until it is handled.
     */
    public static boolean matches(Rule rule, Product product) {
        return switch (rule) {
            case Rule.MinRating r -> product.rating() >= r.value();
            case Rule.MaxPrice r -> product.price().compareTo(r.value()) <= 0;
            case Rule.OfKind r -> r.kind().isInstance(product);
            case Rule.VegetarianOnly r -> !(product instanceof Grocery g) || g.vegetarian();
            case Rule.Not(Rule inner) -> !matches(inner, product);                       // a record pattern
            case Rule.AllOf(var rules) -> rules.stream().allMatch(r -> matches(r, product));
            case Rule.AnyOf(var rules) -> rules.stream().anyMatch(r -> matches(r, product));
        };
    }

    /** A sentence describing the rule, built the same way. */
    public static String describe(Rule rule) {
        return switch (rule) {
            case Rule.MinRating r -> "rated at least " + String.format(Locale.ROOT, "%.1f", r.value());
            case Rule.MaxPrice r -> "costs no more than " + r.value();
            case Rule.OfKind r -> "is a " + r.kind().getSimpleName().toLowerCase();
            case Rule.VegetarianOnly r -> "is vegetarian when it is food";
            case Rule.Not(Rule inner) -> "is not (" + describe(inner) + ")";
            case Rule.AllOf(var rules) -> rules.stream().map(RuleEngine::describe).collect(Collectors.joining(" and ", "(", ")"));
            case Rule.AnyOf(var rules) -> rules.stream().map(RuleEngine::describe).collect(Collectors.joining(" or ", "(", ")"));
        };
    }
}
src/main/java/com/codechanakya/recommend/Recommender.java
package com.codechanakya.recommend;

import java.util.Comparator;
import java.util.List;
import java.util.Optional;

/** Filters a catalogue by rules, scores what is left, and explains each choice. */
public class Recommender {

    private final List<Product> catalogue;

    public Recommender(List<Product> catalogue) {
        this.catalogue = catalogue;
    }

    /** The shopper's best matches, highest score first, ties broken by name. */
    public List<Recommendation> recommend(Shopper shopper, int limit) {
        Rule rule = shopper.toRule();
        return catalogue.stream()
                .filter(product -> RuleEngine.matches(rule, product))
                .map(product -> new Recommendation(product, score(product), highlight(product)))
                .sorted(Comparator.comparingDouble(Recommendation::score).reversed()
                        .thenComparing(r -> r.product().name()))
                .limit(limit)
                .toList();
    }

    /** The single best match, if anything matched at all. */
    public Optional<Recommendation> bestPick(Shopper shopper) {
        return recommend(shopper, 1).stream().findFirst();
    }

    /** Products that match a rule, in catalogue order. */
    public List<Product> filter(Rule rule) {
        return catalogue.stream().filter(product -> RuleEngine.matches(rule, product)).toList();
    }

    /** Rating counts most; each kind of product adds its own small bonus or penalty. */
    static double score(Product product) {
        double base = product.rating() * 10;
        return switch (product) {
            case Book b -> base + (b.pages() > 300 ? 2 : 0);
            case Gadget g -> base + g.warrantyMonths() / 12.0;
            case Grocery g -> base - (g.shelfLifeDays() <= 3 ? 5 : 0);
        };
    }

    /** One sentence about a product. "when" adds a condition to a case, and the first match wins. */
    static String highlight(Product product) {
        return switch (product) {
            case Book b when b.pages() > 500 -> "a long read by " + b.author();
            case Book b -> "a quick read by " + b.author();
            case Gadget g when g.wireless() -> "wireless, with " + g.warrantyMonths() + " months of warranty";
            case Gadget g -> "wired, with " + g.warrantyMonths() + " months of warranty";
            case Grocery(String name, var price, var rating, boolean veg, int days) when days <= 7 ->
                    (veg ? "vegetarian" : "contains meat") + ", fresh: use within " + days + " days";
            case Grocery g -> (g.vegetarian() ? "vegetarian" : "contains meat") + ", keeps for " + g.shelfLifeDays() + " days";
        };
    }
}

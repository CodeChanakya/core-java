Electricity Bill Generator

Beginner if / else Constants Methods Two classes Java 21 or newer

The problem

A utility charges by slab: 3.50 a unit up to 100 units, 5.25 up to 200, 7.80 beyond. Work out the bill for any number of units.

It is the first program with two classes and the first where a method returns a value.

Read in this order

  1. 1BillCalculator.javaThe rates as constants, then billFor and slabFor.
  2. 2Main.javaA table of readings and their bills.
  3. 3expected-output.txtCheck 101 units by hand: 355.25.

The code

src/main/java/BillCalculator.java
import java.math.BigDecimal;

/**
 * Works out an electricity bill from the units consumed in a month.
 *
 * The tariff is split into slabs. Each rate applies only to the units that
 * fall inside its slab, not to the whole consumption.
 */
public final class BillCalculator {

    private static final BigDecimal RATE_UP_TO_100 = new BigDecimal("3.50");
    private static final BigDecimal RATE_101_TO_200 = new BigDecimal("5.25");
    private static final BigDecimal RATE_ABOVE_200 = new BigDecimal("7.80");

    /**
     * Returns the bill for the given units. Negative units make no sense, so
     * they are treated as zero. Phase 4 shows how to reject them properly.
     */
    public static BigDecimal billFor(int units) {
        BigDecimal total = BigDecimal.ZERO;
        int remaining = Math.max(units, 0);

        // Work from the top slab down: bill what is above 200, then cut the
        // remaining units back to 200 and repeat for the next slab.
        if (remaining > 200) {
            total = total.add(RATE_ABOVE_200.multiply(BigDecimal.valueOf(remaining - 200)));
            remaining = 200;
        }
        if (remaining > 100) {
            total = total.add(RATE_101_TO_200.multiply(BigDecimal.valueOf(remaining - 100)));
            remaining = 100;
        }
        total = total.add(RATE_UP_TO_100.multiply(BigDecimal.valueOf(remaining)));
        return total;
    }

    /** Names the highest slab the given units reach. */
    public static String slabFor(int units) {
        if (units <= 100) {
            return "slab 1";
        } else if (units <= 200) {
            return "slab 2";
        } else {
            return "slab 3";
        }
    }
}
src/main/java/Main.java
/**
 * Prints a small bill table for a few sample readings.
 * Read BillCalculator.java first, then this file.
 */
public class Main {

    public static void main(String[] args) {
        System.out.println("Units  Slab    Bill");
        printBill(0);
        printBill(50);
        printBill(100);
        printBill(101);
        printBill(150);
        printBill(250);
    }

    // The edge values 100 and 101 are included on purpose: bugs hide at the
    // boundary between two slabs.
    private static void printBill(int units) {
        System.out.println(units + "  " + BillCalculator.slabFor(units) + "  " + BillCalculator.billFor(units));
    }
}

E-commerce Order Management

Intermediate to advanced Relationships Services Phase project Java 21 or newer

The problem

An online shop sells physical goods and downloads. An order must check stock, work out discount, shipping and GST, and move through payment and dispatch in order.

This is the Phase 3 project. It combines encapsulation, an abstract Product with two subclasses, composition and a service class that makes ordering all or nothing.

package com.codechanakya.shop;

import java.math.BigDecimal;
import java.math.RoundingMode;

/**
 * Something the shop sells. Abstract, because every real product is either
 * physical or digital, and the two ship very differently.
 */
public abstract class Product {

    private final String sku;
    private final String name;
    private final BigDecimal price;
    private int stock;

    protected Product(String sku, String name, BigDecimal price, int stock) {
        this.sku = sku;
        this.name = name;
        this.price = price.setScale(2, RoundingMode.HALF_UP);
        this.stock = stock;
    }

    /** Shipping charged for each unit of this product. */
    public abstract BigDecimal shippingPerUnit();

    public abstract String kind();

    public boolean hasStock(int quantity) {
        return quantity > 0 && quantity <= stock;
    }

    /** Takes units off the shelf. The caller must have checked hasStock first. */
    public void removeStock(int quantity) {
        if (!hasStock(quantity)) {
            throw new IllegalStateException("Not enough stock of " + name);
        }
        stock -= quantity;
    }

    public String getSku() {
        return sku;
    }

    public String getName() {
        return name;
    }

    public BigDecimal getPrice() {
        return price;
    }

    public int getStock() {
        return stock;
    }
}
src/main/java/com/codechanakya/shop/Order.java
package com.codechanakya.shop;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;

/**
 * A customer's order: who bought, what they bought, and where it stands.
 *
 * Relationships used here:
 *   Order -> Customer    association  (the customer outlives the order)
 *   Order -> OrderItem   composition  (the items belong to this order alone)
 *   OrderItem -> Product association  (products live on in the catalog)
 */
public class Order {

    // Plain text statuses for now. Program 14 replaces them with an enum, which
    // is safer because a typo such as "PAYED" would then fail to compile.
    public static final String NEW = "NEW";
    public static final String PAID = "PAID";
    public static final String SHIPPED = "SHIPPED";

    private static final BigDecimal GST_RATE = new BigDecimal("0.18");

    private final String id;
    private final Customer customer;
    private final ArrayList<OrderItem> items = new ArrayList<>();
    private String status = NEW;

    public Order(String id, Customer customer) {
        this.id = id;
        this.customer = customer;
    }

    public void addItem(Product product, int quantity) {
        items.add(new OrderItem(product, quantity));
    }

    public BigDecimal subtotal() {
        BigDecimal sum = BigDecimal.ZERO.setScale(2);
        for (OrderItem item : items) {
            sum = sum.add(item.lineTotal());
        }
        return sum;
    }

    public BigDecimal discount() {
        return subtotal().multiply(customer.discountRate()).setScale(2, RoundingMode.HALF_UP);
    }

    public BigDecimal shipping() {
        BigDecimal sum = BigDecimal.ZERO.setScale(2);
        for (OrderItem item : items) {
            sum = sum.add(item.shipping());
        }
        return sum;
    }

    /** GST is charged on the goods after the discount, not on shipping. */
    public BigDecimal gst() {
        return subtotal().subtract(discount()).multiply(GST_RATE).setScale(2, RoundingMode.HALF_UP);
    }

    public BigDecimal total() {
        return subtotal().subtract(discount()).add(shipping()).add(gst());
    }

    /** NEW to PAID. Returns false if the order is not waiting for payment. */
    public boolean markPaid() {
        if (!status.equals(NEW)) {
            return false;
        }
        status = PAID;
        return true;
    }

    /** PAID to SHIPPED. An unpaid order must not leave the warehouse. */
    public boolean markShipped() {
        if (!status.equals(PAID)) {
            return false;
        }
        status = SHIPPED;
        return true;
    }

    public String invoice() {
        StringBuilder text = new StringBuilder();
        text.append("Order ").append(id).append(" for ").append(customer.getName());
        text.append(customer.isLoyaltyMember() ? " (loyalty member)" : "").append('\n');
        for (OrderItem item : items) {
            text.append(String.format("  %-22s %-8s %2d x %9s = %10s%n",
                    item.getProduct().getName(), item.getProduct().kind(), item.getQuantity(),
                    item.getProduct().getPrice().toPlainString(), item.lineTotal().toPlainString()));
        }
        text.append(String.format("  %-45s %10s%n", "Subtotal", subtotal().toPlainString()));
        text.append(String.format("  %-45s %10s%n", "Loyalty discount", discountText()));
        text.append(String.format("  %-45s %10s%n", "Shipping", shipping().toPlainString()));
        text.append(String.format("  %-45s %10s%n", "GST 18%", gst().toPlainString()));
        text.append(String.format("  %-45s %10s%n", "TOTAL", total().toPlainString()));
        text.append("  Status: ").append(status);
        return text.toString();
    }

    /** A zero discount prints as 0.00, not -0.00. */
    private String discountText() {
        return discount().signum() == 0 ? "0.00" : "-" + discount().toPlainString();
    }

    public String getId() {
        return id;
    }

    public Customer getCustomer() {
        return customer;
    }

    public String getStatus() {
        return status;
    }

    public int itemCount() {
        return items.size();
    }
}
src/main/java/com/codechanakya/shop/OrderService.java
package com.codechanakya.shop;

/**
 * The rules for placing an order. Order and Catalog hold data; this class
 * coordinates them, so a rule like "all or nothing" has one clear home.
 */
public class OrderService {

    private final Catalog catalog;
    private int nextOrderNumber = 5001;

    public OrderService(Catalog catalog) {
        this.catalog = catalog;
    }

    /**
     * Builds an order from parallel arrays of SKUs and quantities.
     *
     * It works in two passes. The first only checks, and throws before anything
     * is touched. The second takes stock and builds the order. That way a
     * failure on the third line cannot leave the first two lines already
     * removed from the shelf.
     */
    public Order placeOrder(Customer customer, String[] skus, int[] quantities) {
        for (int i = 0; i < skus.length; i++) {
            Product product = catalog.find(skus[i]);
            if (product == null) {
                throw new IllegalStateException("Unknown product " + skus[i]);
            }
            if (!product.hasStock(quantities[i])) {
                throw new IllegalStateException(
                        "Only " + product.getStock() + " left of " + product.getName()
                        + " (you asked for " + quantities[i] + ")");
            }
        }

        Order order = new Order("ORD-" + nextOrderNumber, customer);
        nextOrderNumber++;
        for (int i = 0; i < skus.length; i++) {
            Product product = catalog.find(skus[i]);
            product.removeStock(quantities[i]);
            order.addItem(product, quantities[i]);
        }
        return order;
    }
}

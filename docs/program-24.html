Log File Analyzer

Intermediate NIO Path Files Glob Java 21 or newer

The problem

A service writes one log file a day. Someone must answer how many errors there were, which component complains most and where the first error is, from plain text with some lines that are not events.

Path describes locations, Files does the work and throws precise exceptions, and a glob finds only the .log files.

The code

src/main/java/com/codechanakya/logs/LogParser.java
package com.codechanakya.logs;

/**
 * Turns one line of text into a LogEntry.
 *
 * Expected layout: date, time, level, component, then the rest of the line as the message.
 * Columns are separated by one or more spaces, which is why the split uses a pattern.
 */
public final class LogParser {

    private LogParser() {
    }

    /** Returns the entry, or null when the line is not a log line (a stack trace, a blank line, a note). */
    public static LogEntry parse(String line) {
        String[] parts = line.trim().split("\\s+", 5);     // 5 = at most five pieces; the message keeps its spaces
        if (parts.length < 5) {
            return null;
        }
        try {
            Level level = Level.valueOf(parts[2]);
            return new LogEntry(parts[0], parts[1], level, parts[3], parts[4]);
        } catch (IllegalArgumentException e) {
            return null;                                    // third word was not a known level, such as TRACE
        }
    }
}
src/main/java/com/codechanakya/logs/LogFolder.java
package com.codechanakya.logs;

import java.io.IOException;
import java.nio.file.DirectoryStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/** Finds log files in a folder using java.nio.file. */
public final class LogFolder {

    private LogFolder() {
    }

    /**
     * Every *.log file directly inside the folder, sorted by name.
     * The sort matters: the operating system may list a folder in any order, so
     * output that depends on it would differ from one machine to the next.
     */
    public static List<Path> listLogs(Path folder) throws IOException {
        List<Path> found = new ArrayList<>();
        // The second argument is a glob pattern. DirectoryStream holds an open handle, so close it.
        try (DirectoryStream<Path> stream = Files.newDirectoryStream(folder, "*.log")) {
            for (Path path : stream) {
                found.add(path);
            }
        }
        Collections.sort(found);
        return found;
    }
}

Plugin Discovery Demonstration

Advanced Annotations Reflection Phase project Java 21 or newer

The problem

An application should gain behaviour by adding one class, with no edit to the main code.

This Phase 4 project reads class names from a file, inspects them with reflection, checks the @PluginInfo annotation and runs the plugins in phase and order.

Read in this order

  1. 1PluginInfo.javaA custom annotation with defaults.
  2. 2PluginLoader.javaThe questions asked of each class.
  3. 3LoadedPlugin.javaComparable, giving the run order.
  4. 4Main.javaLoad, run and reflect.

The code

src/main/java/com/codechanakya/plugins/PluginInfo.java
package com.codechanakya.plugins;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

/**
 * Labels a class as a plugin and describes it.
 *
 * An annotation is metadata: information ABOUT a class that does not change
 * what the class does by itself. Someone has to read it. Here the PluginLoader
 * reads it at run time using reflection.
 *
 * The two lines above the declaration are annotations on the annotation:
 *   Retention RUNTIME  keep this information in the compiled class so the
 *                      running program can read it (the default would drop it)
 *   Target TYPE        it may only be placed on a class, interface or enum
 */
@Retention(RetentionPolicy.RUNTIME)
@Target(ElementType.TYPE)
public @interface PluginInfo {

    /** The name shown in reports. Required: it has no default. */
    String name();

    /** Smaller numbers run first within a phase. */
    int order() default 100;

    PluginPhase phase() default PluginPhase.MAIN;

    /** A disabled plugin is found but never started. */
    boolean enabled() default true;
}
src/main/java/com/codechanakya/plugins/PluginLoader.java
package com.codechanakya.plugins;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/**
 * Finds plugins by reading class names from a text file and inspecting them.
 *
 * This is reflection: the program looks at classes it did not know about when
 * it was compiled. Nothing here mentions WelcomePlugin or ReportPlugin by
 * name. Add a class and a line in plugins.txt, and it is picked up.
 *
 * Every class goes through the same four questions, and the answer to each is
 * recorded in "skipped" so a reader can see why something did not run.
 */
public class PluginLoader {

    private final List<String> skipped = new ArrayList<>();

    /** Reads the file and returns the plugins that passed, in running order. */
    public List<LoadedPlugin> load(String resourceName) throws IOException, PluginLoadException {
        List<LoadedPlugin> found = new ArrayList<>();

        for (String className : readClassNames(resourceName)) {
            Class<?> type;
            try {
                type = Class.forName(className);                        // 1. does the class exist?
            } catch (ClassNotFoundException e) {
                skipped.add(shortName(className) + ": class not found");
                continue;
            }

            PluginInfo info = type.getAnnotation(PluginInfo.class);     // 2. is it marked @PluginInfo?
            if (info == null) {
                skipped.add(type.getSimpleName() + ": no @PluginInfo annotation");
                continue;
            }
            if (!Plugin.class.isAssignableFrom(type)) {                 // 3. can it run?
                skipped.add(type.getSimpleName() + ": does not implement Plugin");
                continue;
            }
            if (!info.enabled()) {                                      // 4. is it switched on?
                skipped.add(type.getSimpleName() + ": disabled");
                continue;
            }

            try {
                Plugin instance = (Plugin) type.getDeclaredConstructor().newInstance();
                found.add(new LoadedPlugin(instance, info));
            } catch (ReflectiveOperationException e) {
                throw new PluginLoadException("Could not create " + type.getName(), e);
            }
        }

        Collections.sort(found);
        return found;
    }

    public List<String> getSkipped() {
        return skipped;
    }

    private List<String> readClassNames(String resourceName) throws IOException {
        List<String> names = new ArrayList<>();
        try (InputStream in = PluginLoader.class.getResourceAsStream(resourceName)) {
            if (in == null) {
                throw new IOException("Resource not found: " + resourceName);
            }
            BufferedReader reader = new BufferedReader(new InputStreamReader(in, StandardCharsets.UTF_8));
            String line;
            while ((line = reader.readLine()) != null) {
                String trimmed = line.trim();
                if (!trimmed.isEmpty() && !trimmed.startsWith("#")) {
                    names.add(trimmed);
                }
            }
        }
        return names;
    }

    private String shortName(String fullName) {
        return fullName.substring(fullName.lastIndexOf('.') + 1);
    }
}

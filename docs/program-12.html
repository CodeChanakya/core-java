Payment Processing System

Intermediate to advanced Interfaces Composition Immutability Java 21 or newer

The problem

A shop takes payments by card, UPI and wallet, each with its own limits, plus a fee that follows its own rules.

The processor has a payment method and has a fee policy, both typed as interfaces, so any combination works and new ones are added without touching it.

Read in this order

  1. 1PaymentMethod.java, FeePolicy.javaThe two contracts.
  2. 2CardPayment.java, UpiPayment.java, WalletPayment.javaThree ways to pay.
  3. 3PaymentProcessor.javaComposed of the parts above.
  4. 4PaymentResult.javaImmutable outcome.

src/main/java/com/codechanakya/payments/PaymentMethod.java
package com.codechanakya.payments;

import java.math.BigDecimal;

/**
 * Anything that can take a payment: a card, a UPI id, a wallet.
 *
 * The processor only ever sees this interface. It does not know, or care,
 * which concrete class is behind it, so a new way to pay is a new class and
 * the processor does not change.
 */
public interface PaymentMethod {

    /** A short label for receipts, such as "Card ****4242". */
    String name();

    /** Tries to take this amount. Never throws for a normal refusal; it returns a declined result. */
    PaymentResult charge(BigDecimal amount);
}
src/main/java/com/codechanakya/payments/CardPayment.java
package com.codechanakya.payments;

import java.math.BigDecimal;

/** A credit card with a spending limit that shrinks as it is used. */
public class CardPayment implements PaymentMethod {

    private final String last4;
    private BigDecimal remainingLimit;
    private int chargeCount = 0;

    public CardPayment(String last4, BigDecimal limit) {
        this.last4 = last4;
        this.remainingLimit = limit;
    }

    @Override
    public String name() {
        return "Card ****" + last4;
    }

    @Override
    public PaymentResult charge(BigDecimal amount) {
        if (amount.compareTo(remainingLimit) > 0) {
            return PaymentResult.declined("Over the card limit (" + remainingLimit.toPlainString() + " left)");
        }
        remainingLimit = remainingLimit.subtract(amount);
        chargeCount++;
        return PaymentResult.approved(String.format("CARD-%04d", chargeCount));
    }

    public BigDecimal getRemainingLimit() {
        return remainingLimit;
    }
}
src/main/java/com/codechanakya/payments/PaymentProcessor.java
package com.codechanakya.payments;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;

/**
 * Takes payments using whichever method and fee policy it was given.
 *
 * This is composition: the processor HAS a PaymentMethod and HAS a FeePolicy,
 * instead of BEING a card processor or a UPI processor. We could have written
 * CardProcessor, UpiProcessor, WalletProcessor and every combination with
 * every fee rule, and ended up with a pile of near-identical classes. With
 * composition there is one processor, and the parts plug in.
 */
public class PaymentProcessor {

    private PaymentMethod method;
    private FeePolicy feePolicy;
    private final ArrayList<String> ledger = new ArrayList<>();

    public PaymentProcessor(PaymentMethod method, FeePolicy feePolicy) {
        this.method = method;
        this.feePolicy = feePolicy;
    }

    /** Swap how customers pay, without touching anything else. */
    public void useMethod(PaymentMethod method) {
        this.method = method;
    }

    public void useFeePolicy(FeePolicy feePolicy) {
        this.feePolicy = feePolicy;
    }

    /** Charges the amount plus the fee, and records the outcome in the ledger. */
    public PaymentResult pay(BigDecimal amount) {
        if (amount.signum() <= 0) {
            PaymentResult refused = PaymentResult.declined("Amount must be positive");
            ledger.add("DECLINED  " + method.name() + ": " + refused.getMessage());
            return refused;
        }
        BigDecimal fee = feePolicy.feeFor(amount);
        BigDecimal total = amount.add(fee);
        PaymentResult result = method.charge(total);

        if (result.isApproved()) {
            ledger.add(String.format("APPROVED %s charged %s (amount %s + %s) via %s",
                    result.getReference(), total.toPlainString(), amount.setScale(2, RoundingMode.HALF_UP).toPlainString(),
                    feePolicy.describe(), method.name()));
        } else {
            ledger.add("DECLINED  " + method.name() + ": " + result.getMessage());
        }
        return result;
    }

    public void printLedger() {
        System.out.println("--- Ledger ---");
        for (String line : ledger) {
            System.out.println(line);
        }
    }
}

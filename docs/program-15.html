Banking Exception Management

Intermediate Exceptions try / catch Custom exceptions try-with-resources Java 21 or newer

The problem

Returning false says no but not why, and a caller can forget to look. An exception carries the reason and stops the program unless someone deals with it.

This program adds checked and unchecked custom exceptions, chaining, a multi-catch and an audit log that always closes.

The whole program as one picture. Read it top to bottom; the colour of each card says what kind of step it is.

If you are new to coding

Instead of saying only no, the bank says why, and nobody can ignore the answer.

If you are experienced

Checked exceptions force a decision at the boundary; unchecked ones mark programmer errors. try-with-resources guarantees cleanup.

Text version of the flow
  1. A request (input): Deposit, withdraw or transfer.
  2. Is the amount valid? (check): InvalidAmountException (unchecked) for nonsense input.
  3. Is there enough money? (check): InsufficientFundsException (checked) carries the numbers.
  4. Handle or wrap (work): Multi-catch handles it; a wrapped cause keeps the story.
  5. Audit log, always closed (result): AuditLog is AutoCloseable, so try-with-resources closes it.

The code

src/main/java/com/codechanakya/bank/InsufficientFundsException.java
package com.codechanakya.bank;

import java.math.BigDecimal;

/**
 * Thrown when a withdrawal is larger than the balance.
 *
 * This one is checked (it extends Exception). Running out of money is a normal
 * business situation, not a bug, so Java makes every caller decide what to do
 * about it: catch it, or declare it with throws. It also carries data, so the
 * handler can say exactly how much is missing.
 */
public class InsufficientFundsException extends Exception {

    private final BigDecimal requested;
    private final BigDecimal available;

    public InsufficientFundsException(BigDecimal requested, BigDecimal available) {
        super("Cannot withdraw " + requested.toPlainString() + ", balance is " + available.toPlainString());
        this.requested = requested;
        this.available = available;
    }

    public BigDecimal getShortfall() {
        return requested.subtract(available);
    }
}
src/main/java/com/codechanakya/bank/Account.java
package com.codechanakya.bank;

import java.math.BigDecimal;
import java.math.RoundingMode;

/**
 * Program 8's account, rebuilt to report problems with exceptions instead of
 * returning true or false.
 *
 * A boolean says "no" but not why, and a caller can ignore it. An exception
 * carries the reason and cannot be ignored by accident: unhandled, it stops
 * the program.
 */
public class Account {

    private final String accountNumber;
    private final String holderName;
    private BigDecimal balance;

    public Account(String accountNumber, String holderName, BigDecimal openingBalance) {
        this.accountNumber = accountNumber;
        this.holderName = holderName;
        this.balance = openingBalance.setScale(2, RoundingMode.HALF_UP);
    }

    /** Adds money. Throws InvalidAmountException for zero or less. */
    public void deposit(BigDecimal amount) {
        requirePositive(amount);
        balance = balance.add(amount.setScale(2, RoundingMode.HALF_UP));
    }

    /**
     * Takes money out.
     *
     * The "throws" on the first line is a promise to the reader: this method
     * can end with an InsufficientFundsException. Callers must handle it.
     * InvalidAmountException needs no such promise because it is unchecked.
     */
    public void withdraw(BigDecimal amount) throws InsufficientFundsException {
        requirePositive(amount);
        BigDecimal exact = amount.setScale(2, RoundingMode.HALF_UP);
        if (exact.compareTo(balance) > 0) {
            throw new InsufficientFundsException(exact, balance);
        }
        balance = balance.subtract(exact);
    }

    private void requirePositive(BigDecimal amount) {
        if (amount.signum() <= 0) {
            throw new InvalidAmountException(amount);
        }
    }

    public String getAccountNumber() {
        return accountNumber;
    }

    public String getHolderName() {
        return holderName;
    }

    public BigDecimal getBalance() {
        return balance;
    }
}
src/main/java/com/codechanakya/bank/Bank.java
package com.codechanakya.bank;

import java.math.BigDecimal;
import java.util.ArrayList;

public class Bank {

    private final ArrayList<Account> accounts = new ArrayList<>();
    private int nextNumber = 1001;

    public Account openAccount(String holderName, BigDecimal openingBalance) {
        Account account = new Account("ACC-" + nextNumber, holderName, openingBalance);
        nextNumber++;
        accounts.add(account);
        return account;
    }

    /** Program 8 returned null here and hoped callers would check. Now a missing account cannot be missed. */
    public Account find(String accountNumber) throws AccountNotFoundException {
        for (Account account : accounts) {
            if (account.getAccountNumber().equals(accountNumber)) {
                return account;
            }
        }
        throw new AccountNotFoundException(accountNumber);
    }

    /**
     * Moves money between accounts.
     *
     * The two lookups and the withdrawal can each fail in their own way. The
     * multi-catch handles both checked exceptions with one block and wraps them
     * in a single TransferFailedException. Nothing has moved when it is thrown,
     * because the withdrawal comes first and the deposit cannot fail afterwards.
     */
    public void transfer(String fromNumber, String toNumber, BigDecimal amount) throws TransferFailedException {
        try {
            Account from = find(fromNumber);
            Account to = find(toNumber);
            from.withdraw(amount);
            to.deposit(amount);
        } catch (AccountNotFoundException | InsufficientFundsException e) {
            throw new TransferFailedException("Transfer of " + amount.toPlainString() + " failed", e);
        }
    }
}

Vehicle Rental System

Intermediate Abstract classes Interfaces Default methods Java 21 or newer

The problem

An agency rents cars, bikes and trucks, each priced differently, with insurance on some of them.

An abstract class shares the pricing rules and leaves the rate to each subclass. An interface describes the ability to be insured, which a car and a truck have and a bike does not.

Text version of the flow
  1. Build the fleet (input): Car, Motorbike and Truck extend the abstract Vehicle.
  2. Price by days (work): rentalCost is written once and uses each vehicle's daily rate.
  3. Is it insurable? (check): instanceof Insurable adds the insurance line.
  4. Discount for long rentals (work): The rule sits in one place.
  5. Quotes and a refused rental (result): The fleet list, three quotes and one refusal.

The code

src/main/java/com/codechanakya/rental/Vehicle.java
package com.codechanakya.rental;

import java.math.BigDecimal;
import java.math.RoundingMode;

/**
 * Anything the agency can rent out.
 *
 * The class is abstract because "a vehicle" on its own does not exist: you
 * rent a car, a bike or a truck. It still holds everything the types share
 * (registration, model, the long-rental discount) and leaves the parts that
 * differ, the daily rate and the kind, for each subclass to fill in.
 */
public abstract class Vehicle {

    /** Rentals of this many days or more get the long-rental discount. */
    private static final int LONG_RENTAL_DAYS = 7;
    private static final BigDecimal LONG_RENTAL_FACTOR = new BigDecimal("0.90");

    private final String registration;
    private final String model;

    protected Vehicle(String registration, String model) {
        this.registration = registration;
        this.model = model;
    }

    /** What this vehicle costs per day before any discount. Every subclass must supply it. */
    public abstract BigDecimal dailyRate();

    /** A one-word description such as "Car" or "Truck". */
    public abstract String kind();

    /**
     * The rent for a number of days. Written once here, so a long-rental
     * discount applies to every kind of vehicle in exactly the same way.
     */
    public BigDecimal rentalCost(int days) {
        if (days < 1) {
            throw new IllegalArgumentException("A rental must be at least 1 day");
        }
        BigDecimal rent = dailyRate().multiply(BigDecimal.valueOf(days));
        if (days >= LONG_RENTAL_DAYS) {
            rent = rent.multiply(LONG_RENTAL_FACTOR);
        }
        return rent.setScale(2, RoundingMode.HALF_UP);
    }

    public String getRegistration() {
        return registration;
    }

    public String getModel() {
        return model;
    }

    @Override
    public String toString() {
        return String.format("%-10s %-8s %-26s %9s per day", registration, kind(), model, dailyRate().toPlainString());
    }
}
src/main/java/com/codechanakya/rental/Insurable.java
package com.codechanakya.rental;

import java.math.BigDecimal;

/**
 * A promise that a vehicle can be insured.
 *
 * An interface describes what something can do without saying how, and a class
 * may promise several of them. Cars and trucks are insurable; the motorbike is
 * not, and nothing forces it to pretend. That is why this is an interface and
 * not another parent class: Java lets a class extend only one parent.
 */
public interface Insurable {

    BigDecimal insurancePerDay();

    /** A default method: shared behaviour that implementing classes get for free. */
    default BigDecimal insuranceFor(int days) {
        return insurancePerDay().multiply(BigDecimal.valueOf(days));
    }
}
src/main/java/com/codechanakya/rental/Truck.java
package com.codechanakya.rental;

import java.math.BigDecimal;

/** A truck. The daily rate grows with the load it can carry. */
public class Truck extends Vehicle implements Insurable {

    private static final BigDecimal BASE_RATE = new BigDecimal("3000.00");
    private static final BigDecimal PER_TONNE = new BigDecimal("500.00");
    private static final BigDecimal INSURANCE = new BigDecimal("700.00");

    private final int loadTonnes;

    public Truck(String registration, String model, int loadTonnes) {
        super(registration, model);
        this.loadTonnes = loadTonnes;
    }

    @Override
    public BigDecimal dailyRate() {
        return BASE_RATE.add(PER_TONNE.multiply(BigDecimal.valueOf(loadTonnes)));
    }

    @Override
    public String kind() {
        return "Truck";
    }

    @Override
    public BigDecimal insurancePerDay() {
        return INSURANCE;
    }

    public int getLoadTonnes() {
        return loadTonnes;
    }
}
